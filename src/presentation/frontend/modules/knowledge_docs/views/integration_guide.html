# 1. Streaming en tiempo real con SSE (Server-Sent Events) en Python
import httpx
import json

async def ask_contextforge(query: str, project: str = "default"):
    url = f"http://localhost:8000/api/v1/rag/stream?q={query}&project={project}"
    
    async with httpx.AsyncClient(timeout=60.0) as client:
        async with client.stream("GET", url) as response:
            async for line in response.aiter_lines():
                if line.startswith("data:"):
                    data_str = line[5:].strip()
                    if data_str == "[DONE]":
                        break
                    payload = json.loads(data_str)
                    token = payload.get("token", "")
                    sources = payload.get("sources", [])
                    if token:
                        print(token, end="", flush=True)
                    if sources:
                        print(f"\n\nFuentes consultadas: {sources}")
// Consumo reactivo en Frontend con EventSource de HTML5
function connectContextForgeChat(userQuestion, projectSlug = 'default', onToken, onDone) {
  const params = new URLSearchParams({ q: userQuestion, project: projectSlug });
  const sse = new EventSource(`/api/v1/rag/stream?${params.toString()}`);

  sse.addEventListener('token', (event) => {
    try {
      const payload = JSON.parse(event.data);
      const tok = payload.content || payload.token;
      if (tok) onToken(tok);
    } catch (e) {
      onToken(event.data);
    }
  });

  sse.addEventListener('done', () => {
    sse.close();
    if (onDone) onDone();
  });

  sse.onerror = (err) => {
    console.error('SSE Connection error', err);
    sse.close();
  };
}
