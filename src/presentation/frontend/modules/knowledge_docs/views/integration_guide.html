# 1. Streaming en tiempo real con SSE (Server-Sent Events) en Python
import httpx
import json

async def ask_contextforge(query: str, project: str = "default"):
    url = f"http://localhost:8000/api/v1/rag/stream?q={query}&project={project}"
    
    async with httpx.AsyncClient(timeout=60.0) as client:
        async with client.stream("GET", url) as response:
            async for line in response.aiter_lines():
                if line.startswith("data:"):
                    data_str = line[5:].strip()
                    if data_str == "[DONE]":
                        break
                    payload = json.loads(data_str)
                    token = payload.get("token", "")
                    sources = payload.get("sources", [])
                    if token:
                        print(token, end="", flush=True)
                    if sources:
                        print(f"\n\nFuentes consultadas: {sources}")
// Consumo reactivo en Frontend con EventSource de HTML5
function connectContextForgeChat(userQuestion, projectSlug = 'default', onToken, onDone) {
  const params = new URLSearchParams({ q: userQuestion, project: projectSlug });
  const sse = new EventSource(`/api/v1/rag/stream?${params.toString()}`);

  sse.onmessage = (event) => {
    if (event.data === '[DONE]') {
      sse.close();
      if (onDone) onDone();
      return;
    }
    try {
      const payload = JSON.parse(event.data);
      if (payload.token) onToken(payload.token);
    } catch (e) {
      console.warn('Error parsing chunk', e);
    }
  };

  sse.onerror = (err) => {
    console.error('SSE Connection error', err);
    sse.close();
  };
}

Catálogo de Endpoints REST Principales

POST /api/v1/rag/ask

RAG Global Sincrónico

Ejecuta recuperación híbrida (ChromaDB + BM25) y genera una respuesta completa con citas a archivos Markdown.

Body: { query, project_slug, top_k, model? }
GET /api/v1/rag/stream

RAG Streaming en Tiempo Real (SSE)

Emite tokens token-a-token usando Server-Sent Events con baja latencia y citas únicas al finalizar la respuesta.

Query: ?q={pregunta}&project={slug}&top_k={k}
POST /api/v1/chat/scoped

Chat Focalizado (Scoped RAG)

Dialoga con un documento individual y sus anexos multiformato (PDF, Word, Excel) manteniendo historial conversacional.

Body: { library_id, book_id, query, raw_markdown?, history? }
GET /api/v1/search/hybrid

Búsqueda Híbrida Pura

Retorna los fragmentos atómicos más relevantes combinando similitud de coseno y BM25 sin invocar LLM.

Query: ?q={query}&project={slug}&top_k=5
